// 공유해주신 밴드 덕덕 스프레드시트 고유 ID const SPREADSHEET_ID = "1Bjh4G2d4sij45gHZFBn9BNJGjFkXjJAEzTWsrsNcg5Q"; // 배포된 웹앱 URL (프론트엔드 index.html 연동용) // https://script.google.com/macros/s/AKfycbxzKtlhOBmOtmX0CARbUvC_v4coqJmMvgO7lSgYFuqbwomO53LLR-vAPNAwak4k3Gdy/exec // 웹앱으로 POST 요청(데이터 저장)이 들어올 때 실행됩니다. function doPost(e) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('DB'); if (!sheet) { sheet = ss.insertSheet('DB'); } // 기존 데이터 초기화 sheet.clear(); // 카카오톡 대화 내용이 5만자를 넘어가면 구글 시트 1칸 용량을 초과하여 에러가 발생합니다. // 이를 방지하기 위해 데이터를 4만자씩 쪼개어 여러 줄에 걸쳐 저장합니다. (무제한 저장 가능) const payload = e.postData.contents; const chunkSize = 40000; let chunks = []; for (let i = 0; i < payload.length; i += chunkSize) { chunks.push([payload.substring(i, i + chunkSize)]); } // A열에 쪼개진 데이터를 순서대로 기입 sheet.getRange(1, 1, chunks.length, 1).setValues(chunks); return ContentService.createTextOutput(JSON.stringify({ "status": "success" })) .setMimeType(ContentService.MimeType.JSON); } // 웹앱으로 GET 요청(데이터 불러오기)이 들어올 때 실행됩니다. function doGet(e) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('DB'); let data = {}; if (sheet) { const lastRow = sheet.getLastRow(); if (lastRow > 0) { // 쪼개져 저장된 데이터를 모두 가져와 하나로 합칩니다. const values = sheet.getRange(1, 1, lastRow, 1).getValues(); const fullJsonStr = values.map(row => row[0]).join(''); if (fullJsonStr) { try { data = JSON.parse(fullJsonStr); } catch(err) { // JSON 파싱 에러 무시 } } } } // 합쳐진 데이터를 프론트엔드로 전송 return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
